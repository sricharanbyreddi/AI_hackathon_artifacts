\## \*\*Spendid Project – Enterprise Database Design (Finance Operations Command‑Center)\*\* > \*\*Audience:\*\* Enterprise Architects, Data‑Engineers, Security & Compliance leads, Development teams > \*\*Scope:\*\* Core relational data store for the \*\*Brokerage Finance Operations Command Center\*\* (BFOCC). All data is \*\*synthetic\*\* – no real customer, policy, payment or credential data is used. --- ### 1️⃣ Design Principles & Governance | Principle | How it is realised in the design | |-----------|-----------------------------------| | \*\*Security‑by‑Design\*\* | Row‑Level Security (RLS) policies, least‑privilege roles, encrypted columns for PII‑like identifiers (even synthetic), audit columns, immutable audit log. | | \*\*Compliance\*\* | Supports SOX, GDPR‑like data‑subject rights (right to delete/rectify synthetic records), PCI‑DSS‑style segregation of duties (SoD) via role‑based access. | | \*\*Scalability\*\* | Partitioning (range on \`transaction_date\`), sharding‑ready primary keys (UUIDv7), index‑only scans, materialised views for heavy analytics. | | \*\*Auditability & Traceability\*\* | \`audit_log\` table captures every DML change (trigger‑based), \`exception_history\` tracks classification changes, \`copilot_interaction\` logs AI prompts & responses. | | \*\*Governance\*\* | Naming conventions, version‑controlled DDL (Git), change‑management via CI/CD pipelines, data‑dictionary stored in \`metadata.tables\` & \`metadata.columns\`. | | \*\*Performance\*\* | Composite and partial indexes, covering indexes for dashboard queries, connection‑pool sizing, query‑plan hints where needed. | | \*\*Resilience\*\* | Write‑ahead logging (WAL) defaults, hot‑standby replicas, logical replication for analytics DB. | | \*\*Cost‑Efficiency\*\* | Use of \`jsonb\` only for extensible “payload” fields; most data is normalized to avoid duplication. \`\`\`sql -- ========================================================== -- 1. ENUM Types -- ========================================================== CREATE TYPE enum_agency_status AS ENUM ('ACTIVE','INACTIVE'); CREATE TYPE enum_policy_status AS ENUM ('ACTIVE','LAPSED','CANCELLED','EXPIRED'); CREATE TYPE enum_tx_type AS ENUM ('RECEIPT','PAYMENT','COMMISSION','ADJUSTMENT'); CREATE TYPE enum_tx_direction AS ENUM ('IN','OUT'); CREATE TYPE enum_rule_type AS ENUM ('PERCENTAGE','FIXED'); CREATE TYPE enum_commission_status AS ENUM ('PENDING','APPROVED','PAID'); CREATE TYPE enum_journey_name AS ENUM ( 'CASH_TO_CARRIER','AGENCY_BILL','DIRECT_BILL','AR','PAYABLES','COMPENSATION','MONTH_END' ); CREATE TYPE enum_severity AS ENUM ('LOW','MEDIUM','HIGH','CRITICAL'); CREATE TYPE enum_exception_type AS ENUM ('UNAPPLIED_CASH','MISSING_STATEMENT','COMMISSION_SPLIT','AGEING','DUPLICATE_TX','LATE_GL','TECH_ALERT'); CREATE TYPE enum_exception_status AS ENUM ('OPEN','IN_PROGRESS','RESOLVED','CLOSED'); CREATE TYPE enum_operation AS ENUM ('INSERT','UPDATE','DELETE'); CREATE TYPE enum_user_role AS ENUM ('CFO','CONTROLLER','FIN_OPS','ACCOUNTING_MGR','TREASURY','SHARED_SERVICES','AGENT','AUDITOR'); -- ========================================================== -- 2. Core Tables -- ========================================================== CREATE TABLE agency ( agency_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name VARCHAR(200) NOT NULL, legal_entity VARCHAR(200), status enum_agency_status NOT NULL DEFAULT 'ACTIVE', created_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_by UUID NOT NULL, updated_at TIMESTAMPTZ, updated_by UUID, CONSTRAINT uq_agency_name UNIQUE (name) ); CREATE TABLE carrier ( carrier_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name VARCHAR(200) NOT NULL, type VARCHAR(50) NOT NULL, settlement_cutoff TIME NOT NULL, created_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_by UUID NOT NULL, updated_at TIMESTAMPTZ, updated_by UUID, CONSTRAINT uq_carrier_name UNIQUE (name) ); CREATE TABLE producer ( producer_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name VARCHAR(200) NOT NULL, agency_id UUID NOT NULL REFERENCES agency(agency_id) ON DELETE RESTRICT, role VARCHAR(50) NOT NULL, status enum_agency_status NOT NULL DEFAULT 'ACTIVE', created_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_by UUID NOT NULL, updated_at TIMESTAMPTZ, updated_by UUID ); CREATE TABLE policy ( policy_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), policy_number VARCHAR(50) NOT NULL, agency_id UUID NOT NULL REFERENCES agency(agency_id) ON DELETE RESTRICT, carrier_id UUID NOT NULL REFERENCES carrier(carrier_id) ON DELETE RESTRICT, effective_date DATE NOT NULL, expiration_date DATE NOT NULL, premium_amount NUMERIC(15,2) NOT NULL, status enum_policy_status NOT NULL DEFAULT 'ACTIVE', created_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_by UUID NOT NULL, updated_at TIMESTAMPTZ, updated_by UUID, CONSTRAINT uq_policy_number UNIQUE (policy_number) ); CREATE TABLE finance_journey ( journey_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name enum_journey_name NOT NULL, description TEXT, sla_hours INTEGER NOT NULL, created_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_by UUID NOT NULL, updated_at TIMESTAMPTZ, updated_by UUID, CONSTRAINT uq_journey_name UNIQUE (name) ); CREATE TABLE control ( control_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), journey_id UUID NOT NULL REFERENCES finance_journey(journey_id) ON DELETE CASCADE, name VARCHAR(150) NOT NULL, owner_role enum_user_role NOT NULL, cutoff_time TIME, sla_hours INTEGER, is_mandatory BOOLEAN NOT NULL DEFAULT TRUE, created_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_by UUID NOT NULL, updated_at TIMESTAMPTZ, updated_by UUID, CONSTRAINT uq_control_name_per_journey UNIQUE (journey_id, name) ); CREATE TABLE "transaction" ( transaction_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), policy_id UUID NOT NULL REFERENCES policy(policy_id) ON DELETE RESTRICT, type enum_tx_type NOT NULL, direction enum_tx_direction NOT NULL, amount NUMERIC(15,2) NOT NULL CHECK (amount >= 0), currency CHAR(3) NOT NULL DEFAULT 'USD', transaction_date DATE NOT NULL, source_system VARCHAR(100), created_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_by UUID NOT NULL, updated_at TIMESTAMPTZ, updated_by UUID ); CREATE TABLE receipt ( receipt_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), transaction_id UUID NOT NULL REFERENCES "transaction"(transaction_id) ON DELETE RESTRICT, lockbox_file_name VARCHAR(200) NOT NULL, is_applied BOOLEAN NOT NULL DEFAULT FALSE, applied_to_tx_id UUID REFERENCES "transaction"(transaction_id), received_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_by UUID NOT NULL, updated_at TIMESTAMPTZ, updated_by UUID, CONSTRAINT ck_receipt_applied CHECK ( (is_applied = FALSE AND applied_to_tx_id IS NULL) OR (is_applied = TRUE AND applied_to_tx_id IS NOT NULL) ) ); CREATE TABLE carrier_statement ( statement_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), carrier_id UUID NOT NULL REFERENCES carrier(carrier_id) ON DELETE RESTRICT, statement_date DATE NOT NULL, file_name VARCHAR(200) NOT NULL, received_at TIMESTAMPTZ NOT NULL DEFAULT now(), processed BOOLEAN NOT NULL DEFAULT FALSE, created_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_by UUID NOT NULL, updated_at TIMESTAMPTZ, updated_by UUID, CONSTRAINT uq_statement_per_carrier_date UNIQUE (carrier_id, statement_date) ); CREATE TABLE commission_rule ( rule_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), carrier_id UUID NOT NULL REFERENCES carrier(carrier_id) ON DELETE RESTRICT, producer_id UUID NOT NULL REFERENCES producer(producer_id) ON DELETE RESTRICT, rule_type enum_rule_type NOT NULL, value NUMERIC(10,4) NOT NULL CHECK (value >= 0), effective_from DATE NOT NULL, effective_to DATE, is_active BOOLEAN NOT NULL DEFAULT TRUE, created_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_by UUID NOT NULL, updated_at TIMESTAMPTZ, updated_by UUID, CONSTRAINT uq_rule_per_producer UNIQUE (carrier_id, producer_id, effective_from) ); CREATE TABLE commission ( commission_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), transaction_id UUID NOT NULL REFERENCES "transaction"(transaction_id) ON DELETE RESTRICT, rule_id UUID NOT NULL REFERENCES commission_rule(rule_id) ON DELETE RESTRICT, producer_id UUID NOT NULL REFERENCES producer(producer_id) ON DELETE RESTRICT, amount NUMERIC(15,2) NOT NULL CHECK (amount >= 0), status enum_commission_status NOT NULL DEFAULT 'PENDING', calculated_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_at TIMESTAMPTZ NOT NULL DEFAULT now(), created_by UUID NOT NULL, updated_at TIMESTAMPTZ, updated_by UUID );
